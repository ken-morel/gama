1// stb_c_lexer.h - v0.12 - public domain Sean Barrett 2013
2// lexer for making little C-like languages with recursive-descent parsers
3//
4// This file provides both the interface and the implementation.
5// To instantiate the implementation,
6// #define STB_C_LEXER_IMPLEMENTATION
7// in *ONE* source file, before #including this file.
8//
9// The default configuration is fairly close to a C lexer, although
10// suffixes on integer constants are not handled (you can override this).
11//
12// History:
13// 0.12 fix compilation bug for NUL support; better support separate inclusion
14// 0.11 fix clang static analysis warning

18// 0.07 fix mishandling of hexadecimal constants parsed by strtol
19// 0.06 fix missing next character after ending quote mark (Andreas Fredriksson)

23// refactor API to simplify (only one struct instead of two)

282// API function
283void stb_c_lexer_get_location(const stb_lexer *lexer, const char *where, stb_lex_location *loc)
284{
285 char *p = lexer->input_stream;
286 int line_number = 1;
287 int char_offset = 0;
288 while (*p && p < where) {
289 if (*p == '\n' || *p == '\r') {
290 p += (p[0]+p[1] == '\r'+'\n' ? 2 : 1); // skip newline
291 line_number += 1;
292 char_offset = 0;
293 } else {
294 ++p;
295 ++char_offset;
296 }
297 }
298 loc->line_number = line_number;
299 loc->line_offset = char_offset;
300}

332// parse suffixes at the end of a number
333static int stb__clex_parse_suffixes(stb_lexer *lexer, long tokenid, char *start, char *cur, const char *suffixes)
334{
335 #ifdef STB__clex_parse_suffixes
336 lexer->string = lexer->string_storage;
337 lexer->string_len = 0;
338
339 while ((*cur >= 'a' && *cur <= 'z') || (*cur >= 'A' && *cur <= 'Z')) {
340 if (stb__strchr(suffixes, *cur) == 0)
341 return stb__clex_token(lexer, CLEX_parse_error, start, cur);
342 if (lexer->string_len+1 >= lexer->string_storage_len)
343 return stb__clex_token(lexer, CLEX_parse_error, start, cur);
344 lexer->string[lexer->string_len++] = *cur++;
345 }
346 #else
347 suffixes = suffixes; // attempt to suppress warnings
348 #endif
349 return stb__clex_token(lexer, tokenid, start, cur-1);
350}

353static double stb__clex_pow(double base, unsigned int exponent)
354{
355 double value=1;
356 for ( ; exponent; exponent >>= 1) {
357 if (exponent & 1)
358 value *= base;
359 base *= base;
360 }
361 return value;

364static double stb__clex_parse_float(char *p, char **q)
365{
366 char *s = p;
367 double value=0;
368 int base=10;
369 int exponent=0;
370
371#ifdef STB__clex_hex_floats
372 if (*p == '0') {
373 if (p[1] == 'x' || p[1] == 'X') {
374 base=16;
375 p += 2;
376 }
377 }
378#endif
379
380 for (;;) {
381 if (*p >= '0' && *p <= '9')
382 value = value*base + (*p++ - '0');
383#ifdef STB__clex_hex_floats
384 else if (base == 16 && *p >= 'a' && *p <= 'f')
385 value = value*base + 10 + (*p++ - 'a');
386 else if (base == 16 && *p >= 'A' && *p <= 'F')
387 value = value*base + 10 + (*p++ - 'A');
388#endif
389 else
390 break;
391 }
392
393 if (*p == '.') {
394 double pow, addend = 0;
395 ++p;
396 for (pow=1; ; pow*=base) {
397 if (*p >= '0' && *p <= '9')
398 addend = addend*base + (*p++ - '0');
399#ifdef STB__clex_hex_floats
400 else if (base == 16 && *p >= 'a' && *p <= 'f')
401 addend = addend*base + 10 + (*p++ - 'a');
402 else if (base == 16 && *p >= 'A' && *p <= 'F')
403 addend = addend*base + 10 + (*p++ - 'A');
404#endif
405 else
406 break;
407 }
408 value += addend / pow;
409 }
410#ifdef STB__clex_hex_floats
411 if (base == 16) {
412 // exponent required for hex float literal
413 if (*p != 'p' && *p != 'P') {
414 *q = s;
415 return 0;
416 }
417 exponent = 1;
418 } else
419#endif
420 exponent = (*p == 'e' || *p == 'E');
421
422 if (exponent) {
423 int sign = p[1] == '-';
424 unsigned int exponent=0;
425 double power=1;
426 ++p;
427 if (*p == '-' || *p == '+')
428 ++p;
429 while (*p >= '0' && *p <= '9')
430 exponent = exponent*10 + (*p++ - '0');
431
432#ifdef STB__clex_hex_floats
433 if (base == 16)
434 power = stb__clex_pow(2, exponent);
435 else
436#endif
437 power = stb__clex_pow(10, exponent);
438 if (sign)
439 value /= power;
440 else
441 value *= power;
442 }
443 *q = p;
444 return value;
445}
446#endif

448static int stb__clex_parse_char(char *p, char **q)
449{
450 if (*p == '\\') {
451 *q = p+2; // tentatively guess we'll parse two characters
452 switch(p[1]) {
453 case '\\': return '\\';
454 case '\'': return '\'';
455 case '"': return '"';
456 case 't': return '\t';
457 case 'f': return '\f';
458 case 'n': return '\n';
459 case 'r': return '\r';
460 case '0': return '\0'; // @TODO ocatal constants
461 case 'x': case 'X': return -1; // @TODO hex constants
462 case 'u': return -1; // @TODO unicode constants
463 }
464 }
465 *q = p+1;
466 return (unsigned char) *p;
467}

469static int stb__clex_parse_string(stb_lexer *lexer, char *p, int type)
470{
471 char *start = p;
472 char delim = *p++; // grab the " or ' for later matching
473 char *out = lexer->string_storage;
474 char *outend = lexer->string_storage + lexer->string_storage_len;
475 while (*p != delim) {
476 int n;
477 if (*p == '\\') {
478 char *q;
479 n = stb__clex_parse_char(p, &q);
480 if (n < 0)
481 return stb__clex_token(lexer, CLEX_parse_error, start, q);
482 p = q;
483 } else {
484 // @OPTIMIZE: could speed this up by looping-while-not-backslash
485 n = (unsigned char) *p++;
486 }
487 if (out+1 > outend)
488 return stb__clex_token(lexer, CLEX_parse_error, start, p);
489 // @TODO expand unicode escapes to UTF8
490 *out++ = (char) n;
491 }
492 *out = 0;
493 lexer->string = lexer->string_storage;
494 lexer->string_len = (int) (out - lexer->string_storage);
495 return stb__clex_token(lexer, type, start, p);
496}

498int stb_c_lexer_get_token(stb_lexer *lexer)
499{
500 char *p = lexer->parse_point;
501
502 // skip whitespace and comments
503 for (;;) {
504 #ifdef STB_C_LEX_ISWHITE
505 while (p != lexer->stream_end) {
506 int n;
507 n = STB_C_LEX_ISWHITE(p);
508 if (n == 0) break;
509 if (lexer->eof && lexer->eof - lexer->parse_point < n)
510 return stb__clex_token(tok, CLEX_parse_error, p,lexer->eof-1);
511 p += n;
512 }
513 #else
514 while (p != lexer->eof && stb__clex_iswhite(*p))
515 ++p;
516 #endif
517
518 STB_C_LEX_CPP_COMMENTS(
519 if (p != lexer->eof && p[0] == '/' && p[1] == '/') {
520 while (p != lexer->eof && *p != '\r' && *p != '\n')
521 ++p;
522 continue;
523 }
524 )
525
526 STB_C_LEX_C_COMMENTS(
527 if (p != lexer->eof && p[0] == '/' && p[1] == '*') {
528 char *start = p;
529 p += 2;
530 while (p != lexer->eof && (p[0] != '*' || p[1] != '/'))
531 ++p;
532 if (p == lexer->eof)
533 return stb__clex_token(lexer, CLEX_parse_error, start, p-1);
534 p += 2;
535 continue;
536 }
537 )
538
539 #ifdef STB__clex_discard_preprocessor
540 // @TODO this discards everything after a '#', regardless
541 // of where in the line the # is, rather than requiring it
542 // be at the start. (because this parser doesn't otherwise
543 // check for line breaks!)
544 if (p != lexer->eof && p[0] == '#') {
545 while (p != lexer->eof && *p != '\r' && *p != '\n')
546 ++p;
547 continue;
548 }
549 #endif
550
551 break;
552 }
553
554 if (p == lexer->eof)
555 return stb__clex_eof(lexer);
556
557 switch (*p) {
558 default:
559 if ( (*p >= 'a' && *p <= 'z')
560 || (*p >= 'A' && *p <= 'Z')
561 || *p == '_' || (unsigned char) *p >= 128 // >= 128 is UTF8 char
562 STB_C_LEX_DOLLAR_IDENTIFIER( || *p == '$' ) )
563 {
564 int n = 0;
565 lexer->string = lexer->string_storage;
566 do {
567 if (n+1 >= lexer->string_storage_len)
568 return stb__clex_token(lexer, CLEX_parse_error, p, p+n);
569 lexer->string[n] = p[n];
570 ++n;
571 } while (
572 (p[n] >= 'a' && p[n] <= 'z')
573 || (p[n] >= 'A' && p[n] <= 'Z')
574 || (p[n] >= '0' && p[n] <= '9') // allow digits in middle of identifier
575 || p[n] == '_' || (unsigned char) p[n] >= 128
576 STB_C_LEX_DOLLAR_IDENTIFIER( || p[n] == '$' )
577 );
578 lexer->string[n] = 0;
579 lexer->string_len = n;
580 return stb__clex_token(lexer, CLEX_id, p, p+n-1);
581 }
582
583 // check for EOF
584 STB_C_LEX_0_IS_EOF(
585 if (*p == 0)
586 return stb__clex_eof(lexer);
587 )
588
589 single_char:
590 // not an identifier, return the character as itself
591 return stb__clex_token(lexer, *p, p, p);
592
593 case '+':
594 if (p+1 != lexer->eof) {
595 STB_C_LEX_C_INCREMENTS(if (p[1] == '+') return stb__clex_token(lexer, CLEX_plusplus, p,p+1);)
596 STB_C_LEX_C_ARITHEQ( if (p[1] == '=') return stb__clex_token(lexer, CLEX_pluseq , p,p+1);)
597 }
598 goto single_char;
599 case '-':
600 if (p+1 != lexer->eof) {
601 STB_C_LEX_C_INCREMENTS(if (p[1] == '-') return stb__clex_token(lexer, CLEX_minusminus, p,p+1);)
602 STB_C_LEX_C_ARITHEQ( if (p[1] == '=') return stb__clex_token(lexer, CLEX_minuseq , p,p+1);)
603 STB_C_LEX_C_ARROW( if (p[1] == '>') return stb__clex_token(lexer, CLEX_arrow , p,p+1);)
604 }
605 goto single_char;
606 case '&':
607 if (p+1 != lexer->eof) {
