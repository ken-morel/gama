42int gm3_obj_parse_next_line(char **end, gm3ObjLine *ln) {
43 memset(ln, 0, sizeof(*ln));
44 ln->type = '?';
45
46 char *p = gmu_skip_ws(*end);
47 if (*p == '\0')
48 return 0;
49
50 if (p[0] == 'v' && isspace((unsigned char)p[1])) {
51 ln->type = 'v';
52 p = gmu_skip_ws(p + 1);
53 ln->points[0] = strtod(p, &p);
54 ln->points[1] = strtod(p, &p);
55 ln->points[2] = strtod(p, &p);
56 } else if (p[0] == 'v' && p[1] == 'n') {
57 ln->type = 'n';
58 p = gmu_skip_ws(p + 2);
59 ln->points[0] = strtod(p, &p);
60 ln->points[1] = strtod(p, &p);
61 ln->points[2] = strtod(p, &p);
62 } else if (p[0] == 'v' && p[1] == 't') {
63 ln->type = 't';
64 p = gmu_skip_ws(p + 2);
65 ln->points[0] = strtod(p, &p);
66 ln->points[1] = strtod(p, &p);
67 } else if (p[0] == 'f' && isspace((unsigned char)p[1])) {
68 ln->type = 'f';
69 p = gmu_skip_ws(p + 1);
70 while (*p != '\0' && *p != '\n' && *p != '\r' && ln->n_indices < 64) {
71 ln->indices[ln->n_indices][0] = _gm3u_parse_idx(&p);
72 if (*p == '/') {
73 p++;
74 ln->indices[ln->n_indices][1] = _gm3u_parse_idx(&p);
75 if (*p == '/') {
76 p++;
77 ln->indices[ln->n_indices][2] = _gm3u_parse_idx(&p);
78 }
79 }
80 ln->n_indices++;
81 p = gmu_skip_ws(p);
82 }
83 } else if (strncmp(p, "mtllib", 6) == 0) {
84 ln->type = 'L';
85 gm3u_str_copy_eol(ln->name, p + 6, sizeof(ln->name));
86 } else if (strncmp(p, "usemtl", 6) == 0) {
87 ln->type = 'U';
88 gm3u_str_copy_eol(ln->name, p + 6, sizeof(ln->name));
89 } else if (p[0] == 'o' || p[0] == 'g') {
90 ln->type = p[0];
91 gm3u_str_copy_eol(ln->name, p + 1, sizeof(ln->name));
92 } else if (p[0] == '#') {
93 ln->type = '#';
94 }
95
96 while (*p && *p != '\n')
97 p++;
98 if (*p == '\n')
99 p++;
100 *end = p;
101 return 1;
102}

113int gm3_obj_parse(char *content, gm3ObjLine **result, size_t *n_lines) {
114 char *pos = content;
115 size_t buffer_size = 128;
116 size_t buffer_index = 0;
117 gm3ObjLine *buffer = malloc(buffer_size * sizeof(gm3ObjLine));
118 if (!buffer)
119 return -1;
120
121 gm3ObjLine ln;
122 while (gm3_obj_parse_next_line(&pos, &ln)) {
123 if (buffer_index >= buffer_size) {
124 size_t new_size = buffer_size * 2;
125 gm3ObjLine *newbuff = realloc(buffer, new_size * sizeof(gm3ObjLine));
126 if (!newbuff) {
127 free(buffer);
128 return -1;
129 }
130 buffer = newbuff;
131 buffer_size = new_size;
132 }
133 buffer[buffer_index++] = ln;
134 }
135
136 *n_lines = buffer_index;
137 *result = buffer;
138 return 0;
