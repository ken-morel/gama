1// stb_voxel_render.h - v0.89 - Sean Barrett, 2015 - public domain
2//
3// This library helps render large-scale "voxel" worlds for games,
4// in this case, one with blocks that can have textures and that
5// can also be a few shapes other than cubes.
6//
7// Video introduction:
8// http://www.youtube.com/watch?v=2vnTtiLrV1w
9//
10// Minecraft-viewer sample app (not very simple though):
11// http://github.com/nothings/stb/tree/master/tests/caveview
12//
13// It works by creating triangle meshes. The library includes
14//
15// - converter from dense 3D arrays of block info to vertex mesh
16// - vertex & fragment shaders for the vertex mesh
17// - assistance in setting up shader state
18//
19// For portability, none of the library code actually accesses
20// the 3D graphics API. (At the moment, it's not actually portable
21// since the shaders are GLSL only, but patches are welcome.)
22//
23// You have to do all the caching and tracking of vertex buffers
24// yourself. However, you could also try making a game with
25// a small enough world that it's fully loaded rather than
26// streaming. Currently the preferred vertex format is 20 bytes
27// per quad. There are designs to allow much more compact formats
28// with a slight reduction in shader features, but no roadmap
29// for actually implementing them.
30//
31//
32// USAGE
33//
34// #define the symbol STB_VOXEL_RENDER_IMPLEMENTATION in *one*
35// C/C++ file before the #include of this file; the implementation
36// will be generated in that file.
37//
38// If you define the symbols STB_VOXEL_RENDER_STATIC, then the
39// implementation will be private to that file.
40//
41//
42// FEATURES
43//
44// - you can choose textured blocks with the features below,
45// or colored voxels with 2^24 colors and no textures.
46//
47// - voxels are mostly just cubes, but there's support for
48// half-height cubes and diagonal slopes, half-height
49// diagonals, and even odder shapes especially for doing
50// more-continuous "ground".
51//
52// - texture coordinates are projections along one of the major
53// axes, with the per-texture scaling.
54//
55// - a number of aspects of the shader and the vertex format
56// are configurable; the library generally takes care of
57// coordinating the vertex format with the mesh for you.
58//
59//
60// FEATURES (SHADER PERSPECTIVE)
61//
62// - vertices aligned on integer lattice, z on multiples of 0.5
63// - per-vertex "lighting" or "ambient occlusion" value (6 bits)
64// - per-vertex texture crossfade (3 bits)
65//
66// - per-face texture #1 id (8-bit index into array texture)
67// - per-face texture #2 id (8-bit index into second array texture)
68// - per-face color (6-bit palette index, 2 bits of per-texture boolean enable)
69// - per-face 5-bit normal for lighting calculations & texture coord computation
70// - per-face 2-bit texture matrix rotation to rotate faces
71//
72// - indexed-by-texture-id scale factor (separate for texture #1 and texture #2)
73// - indexed-by-texture-#2-id blend mode (alpha composite or modulate/multiply);
74// the first is good for decals, the second for detail textures, "light maps",
75// etc; both modes are controlled by texture #2's alpha, scaled by the
76// per-vertex texture crossfade and the per-face color (if enabled on texture #2);
77// modulate/multiply multiplies by an extra factor of 2.0 so that if you
78// make detail maps whose average brightness is 0.5 everything works nicely.
79//
80// - ambient lighting: half-lambert directional plus constant, all scaled by vertex ao
81// - face can be fullbright (emissive), controlled by per-face color
82// - installable lighting, with default single-point-light
83// - installable fog, with default hacked smoothstep
84//
85// Note that all the variations of lighting selection and texture
86// blending are run-time conditions in the shader, so they can be
87// intermixed in a single mesh.
88//
89//
90// INTEGRATION ARC
91//
92// The way to get this library to work from scratch is to do the following:
93//
94// Step 1. define STBVOX_CONFIG_MODE to 0
95//
96// This mode uses only vertex attributes and uniforms, and is easiest
97// to get working. It requires 32 bytes per quad and limits the
98// size of some tables to avoid hitting uniform limits.
99//
100// Step 2. define STBVOX_CONFIG_MODE to 1
101//
102// This requires using a texture buffer to store the quad data,
103// reducing the size to 20 bytes per quad.
104//
105// Step 3: define STBVOX_CONFIG_PREFER_TEXBUFFER
106//
107// This causes some uniforms to be stored as texture buffers
108// instead. This increases the size of some of those tables,
109// and avoids a potential slow path (gathering non-uniform
110// data from uniforms) on some hardware.
111//
112// In the future I might add additional modes that have significantly
113// smaller meshes but reduce features, down as small as 6 bytes per quad.
114// See elsewhere in this file for a table of candidate modes. Switching
115// to a mode will require changing some of your mesh creation code, but
116// everything else should be seamless. (And I'd like to change the API
117// so that mesh creation is data-driven the way the uniforms are, and
118// then you wouldn't even have to change anything but the mode number.)
119//
120//
121// IMPROVEMENTS FOR SHIP-WORTHY PROGRAMS USING THIS LIBRARY
122//
123// I currently tolerate a certain level of "bugginess" in this library.
124//
125// I'm referring to things which look a little wrong (as long as they
126// don't cause holes or cracks in the output meshes), or things which
127// do not produce as optimal a mesh as possible. Notable examples:
128//
129// - incorrect lighting on slopes
130// - inefficient meshes for vheight blocks
131//
132// I am willing to do the work to improve these things if someone is
133// going to ship a substantial program that would be improved by them.
134// (It need not be commercial, nor need it be a game.) I just didn't
135// want to do the work up front if it might never be leveraged. So just
136// submit a bug report as usual (github is preferred), but add a note
137// that this is for a thing that is really going to ship. (That means
138// you need to be far enough into the project that it's clear you're
139// committed to it; not during early exploratory development.)
140//
141//
142// VOXEL MESH API
143//
144// Context
145//
146// To understand the API, make sure you first understand the feature set
147// listed above.
148//
149// Because the vertices are compact, they have very limited spatial
150// precision. Thus a single mesh can only contain the data for a limited
151// area. To make very large voxel maps, you'll need to build multiple
152// vertex buffers. (But you want this anyway for frustum culling.)
153//
154// Each generated mesh has three components:
155// - vertex data (vertex buffer)
156// - face data (optional, stored in texture buffer)
157// - mesh transform (uniforms)
158//
159// Once you've generated the mesh with this library, it's up to you
160// to upload it to the GPU, to keep track of the state, and to render
161// it.
162//
163// Concept
164//
165// The basic design is that you pass in one or more 3D arrays; each array
166// is (typically) one-byte-per-voxel and contains information about one
167// or more properties of some particular voxel property.
168//
169// Because there is so much per-vertex and per-face data possible
170// in the output, and each voxel can have 6 faces and 8 vertices, it
171// would require an very large data structure to describe all
172// of the possibilities, and this would cause the mesh-creation
173// process to be slow. Instead, the API provides multiple ways
174// to express each property, some more compact, others less so;
175// each such way has some limitations on what it can express.
176//
177// Note that there are so many paths and combinations, not all of them
178// have been tested. Just report bugs and I'll fix 'em.
179//
180// Details
181//
182// See the API documentation in the header-file section.

248// CONFIGURATION MACROS
249//
250// #define STBVOX_CONFIG_MODE <integer> // REQUIRED
251// Configures the overall behavior of stb_voxel_render. This
252// can affect the shaders, the uniform info, and other things.
253// (If you need more than one mode in the same app, you can
254// use STB_VOXEL_RENDER_STATIC to create multiple versions
255// in separate files, and then wrap them.)
256//
257// Mode value Meaning
258// 0 Textured blocks, 32-byte quads
259// 1 Textured blocks, 20-byte quads
260// 20 Untextured blocks, 32-byte quads
261// 21 Untextured blocks, 20-byte quads
262//
263//
264// #define STBVOX_CONFIG_PRECISION_Z <integer> // OPTIONAL
265// Defines the number of bits of fractional position for Z.
266// Only 0 or 1 are valid. 1 is the default. If 0, then a
267// single mesh has twice the legal Z range; e.g. in
268// modes 0,1,20,21, Z in the mesh can extend to 511 instead
269// of 255. However, half-height blocks cannot be used.
270//
271// All of the following are just #ifdef tested so need no values, and are optional.
272//
273// STBVOX_CONFIG_BLOCKTYPE_SHORT
274// use unsigned 16-bit values for 'blocktype' in the input instead of 8-bit values
275//
276// STBVOX_CONFIG_OPENGL_MODELVIEW
277// use the gl_ModelView matrix rather than the explicit uniform
278//
279// STBVOX_CONFIG_HLSL
280// NOT IMPLEMENTED! Define HLSL shaders instead of GLSL shaders
281//
282// STBVOX_CONFIG_PREFER_TEXBUFFER
283// Stores many of the uniform arrays in texture buffers instead,
284// so they can be larger and may be more efficient on some hardware.
285//
286// STBVOX_CONFIG_LIGHTING_SIMPLE
287// Creates a simple lighting engine with a single point light source
288// in addition to the default half-lambert ambient light.
289//
290// STBVOX_CONFIG_LIGHTING
291// Declares a lighting function hook; you must append a lighting function
292// to the shader before compiling it:
293// vec3 compute_lighting(vec3 pos, vec3 norm, vec3 albedo, vec3 ambient);
294// 'ambient' is the half-lambert ambient light with vertex ambient-occlusion applied
295//
296// STBVOX_CONFIG_FOG_SMOOTHSTEP
297// Defines a simple unrealistic fog system designed to maximize
298// unobscured view distance while not looking too weird when things
299// emerge from the fog. Configured using an extra array element
300// in the STBVOX_UNIFORM_ambient uniform.
301//
302// STBVOX_CONFIG_FOG
303// Defines a fog function hook; you must append a fog function to
304// the shader before compiling it:
305// vec3 compute_fog(vec3 color, vec3 relative_pos, float fragment_alpha);
306// "color" is the incoming pre-fogged color, fragment_alpha is the alpha value,
307// and relative_pos is the vector from the point to the camera in worldspace
308//
309// STBVOX_CONFIG_DISABLE_TEX2
310// This disables all processing of texture 2 in the shader in case
311// you don't use it. Eventually this could be replaced with a mode
312// that omits the unused data entirely.
313//
314// STBVOX_CONFIG_TEX1_EDGE_CLAMP
315// STBVOX_CONFIG_TEX2_EDGE_CLAMP
316// If you want to edge clamp the textures, instead of letting them wrap,
317// set this flag. By default stb_voxel_render relies on texture wrapping
318// to simplify texture coordinate generation. This flag forces it to do
319// it correctly, although there can still be minor artifacts.
320//
321// STBVOX_CONFIG_ROTATION_IN_LIGHTING
322// Changes the meaning of the 'lighting' mesher input variable to also
323// store the rotation; see later discussion.
324//
325// STBVOX_CONFIG_VHEIGHT_IN_LIGHTING
326// Changes the meaning of the 'lighting' mesher input variable to also
327// store the vheight; see later discussion. Cannot use both this and
328// the previous variable.
329//
330// STBVOX_CONFIG_PREMULTIPLIED_ALPHA
331// Adjusts the shader calculations on the assumption that tex1.rgba,
332// tex2.rgba, and color.rgba all use premultiplied values, and that
333// the output of the fragment shader should be premultiplied.
334//
335// STBVOX_CONFIG_UNPREMULTIPLY
336// Only meaningful if STBVOX_CONFIG_PREMULTIPLIED_ALPHA is defined.
337// Changes the behavior described above so that the inputs are
338// still premultiplied alpha, but the output of the fragment
339// shader is not premultiplied alpha. This is needed when allowing
340// non-unit alpha values but not doing alpha-blending (for example
341// when alpha testing).
342//

363// Call this to set the buffer into which stbvox will write the mesh
364// it creates. It can build more than one mesh in parallel (distinguished
365// by the 'mesh' parameter), and each mesh can be made up of more than
366// one buffer (distinguished by the 'slot' parameter).
367//

382// be four times as large as the second buffer. The first buffer
383// contains a single vertex attribute: 'attr_vertex', a single 32-bit uint.
384// The second buffer contains texture buffer data (an array of 32-bit uints)
385// that will be accessed through the sampler identified by STBVOX_UNIFORM_face_data.

414STBVXDEC void stbvox_set_input_range(stbvox_mesh_maker *mm, int x0, int y0, int z0, int x1, int y1, int z1);
415// This sets the range of values in the 3D array for the voxels that
416// the mesh generator will convert. The lower values are inclusive,
417// the higher values are exclusive, so (0,0,0) to (16,16,16) generates
418// mesh data associated with voxels up to (15,15,15) but no higher.
419//
420// The mesh generate generates faces at the boundary between open space
421// and solid space but associates them with the solid space, so if (15,0,0)
422// is open and (16,0,0) is solid, then the mesh will contain the boundary
423// between them if x0 <= 16 and x1 > 16.
424//
425// Note that the mesh generator will access array elements 1 beyond the
426// limits set in these parameters. For example, if you set the limits
427// to be (0,0,0) and (16,16,16), then the generator will access all of
428// the voxels between (-1,-1,-1) and (16,16,16), including (16,16,16).
429// You may have to do pointer arithmetic to make it work.
430//
431// For example, caveview processes mesh chunks that are 32x32x16, but it
432// does this using input buffers that are 34x34x18.
433//
434// The lower limits are x0 >= 0, y0 >= 0, and z0 >= 0.
435//
436// The upper limits are mode dependent, but all the current methods are
437// limited to x1 < 127, y1 < 127, z1 < 255. Note that these are not
438// powers of two; if you want to use power-of-two chunks (to make
439// it efficient to decide which chunk a coordinate falls in), you're
440// limited to at most x1=64, y1=64, z1=128. For classic Minecraft-style
441// worlds with limited vertical extent, I recommend using a single
442// chunk for the entire height, which limits the height to 255 blocks
443// (one less than Minecraft), and only chunk the map in X & Y.
444

446// Call this function to create mesh data for the currently configured
447// set of input data. This appends to the currently configured mesh output
448// buffer. Returns 1 on success. If there is not enough room in the buffer,
449// it outputs as much as it can, and returns 0; you need to switch output
450// buffers (either by calling stbvox_set_buffer to set new buffers, or
451// by copying the data out and calling stbvox_reset_buffers), and then

456// 32x32x255, but builds the mesh for it by processing 32x32x16 at atime
457// (this is faster as it is reuses the same 34x34x18 input buffers rather

531STBVXDEC int stbvox_get_uniform_info(stbvox_uniform_info *info, int uniform);
532// Gets the information about a uniform necessary for you to
533// set up each uniform with a minimal amount of explicit code.
534// See the sample code after the structure definition for stbvox_uniform_info,
535// further down in this header section.
536//
537// "uniform" is from the list immediately following. For many
538// of these, default values are provided which you can set.
539// Most values are shared for most draw calls; e.g. for stateful
540// APIs you can set most of the state only once. Only
541// STBVOX_UNIFORM_transform needs to change per draw call.
542//
543// STBVOX_UNIFORM_texscale
544// 64- or 128-long vec4 array. (128 only if STBVOX_CONFIG_PREFER_TEXBUFFER)
545// x: scale factor to apply to texture #1. must be a power of two. 1.0 means 'face-sized'
546// y: scale factor to apply to texture #2. must be a power of two. 1.0 means 'face-sized'
547// z: blend mode indexed by texture #2. 0.0 is alpha compositing; 1.0 is multiplication.
548// w: unused currently. @TODO use to support texture animation?
549//
550// Texscale is indexed by the bottom 6 or 7 bits of the texture id; thus for
551// example the texture at index 0 in the array and the texture in index 128 of
552// the array must be scaled the same. This means that if you only have 64 or 128
553// unique textures, they all get distinct values anyway; otherwise you have
554// to group them in pairs or sets of four.
555//
556// STBVOX_UNIFORM_ambient
557// 4-long vec4 array:
558// ambient[0].xyz - negative of direction of a directional light for half-lambert
559// ambient[1].rgb - color of light scaled by NdotL (can be negative)

1020//
1021// VHEIGHT DOCUMENTATION
1022//
1023// "vheight" is the internal name for the special block types
1024// with sloped tops or bottoms. "vheight" stands for "vertex height".
1025//
1026// Note that these blocks are very flexible (there are 256 of them,
1027// although at least 17 of them should never be used), but they
1028// also have a disadvantage that they generate extra invisible
1029// faces; the generator does not currently detect whether adjacent
1030// vheight blocks hide each others sides, so those side faces are
1031// always generated. For a continuous ground terrain, this means
1032// that you may generate 5x as many quads as needed. See notes
1033// on "improvements for shipping products" in the introduction.
1034
1035enum
1036{
1037 STBVOX_VERTEX_HEIGHT_0,
1038 STBVOX_VERTEX_HEIGHT_half,
1039 STBVOX_VERTEX_HEIGHT_1,
1040 STBVOX_VERTEX_HEIGHT_one_and_a_half,
1041};
1042// These are the "vheight" values. Vheight stands for "vertex height".
1043// The idea is that for a "floor vheight" block, you take a cube and
1044// reposition the top-most vertices at various heights as specified by
1045// the vheight values. Similarly, a "ceiling vheight" block takes a
1046// cube and repositions the bottom-most vertices.
1047//
1048// A floor block only adjusts the top four vertices; the bottom four vertices
1049// remain at the bottom of the block. The height values are 2 bits,
1050// measured in halves of a block; so you can specify heights of 0/2,
1051// 1/2, 2/2, or 3/2. 0 is the bottom of the block, 1 is halfway
1052// up the block, 2 is the top of the block, and 3 is halfway up the
1053// next block (and actually outside of the block). The value 3 is
1054// actually legal for floor vheight (but not ceiling), and allows you to:
1055//
1056// (A) have smoother terrain by having slopes that cross blocks,
1057// e.g. (1,1,3,3) is a regular-seeming slope halfway between blocks
1058// (B) make slopes steeper than 45-degrees, e.g. (0,0,3,3)
1059//
1060// (Because only z coordinates have half-block precision, and x&y are
1061// limited to block corner precision, it's not possible to make these
1062// things "properly" out of blocks, e.g. a half-slope block on its side
1063// or a sloped block halfway between blocks that's made out of two blocks.)
1064//
1065// If you define STBVOX_CONFIG_OPTIMIZED_VHEIGHT, then the top face
1066// (or bottom face for a ceiling vheight block) will be drawn as a
1067// single quad even if the four vertex heights aren't planar, and a
1068// single normal will be used over the entire quad. If you
1069// don't define it, then if the top face is non-planar, it will be
1070// split into two triangles, each with their own normal/lighting.
1071// (Note that since all output from stb_voxel_render is quad meshes,
1072// triangles are actually rendered as degenerate quads.) In this case,
1073// the distinction between STBVOX_GEOM_floor_vheight_03 and
1074// STBVOX_GEOM_floor_vheight_12 comes into play; the former introduces
1075// an edge from the SW to NE corner (i.e. from <0,0,?> to <1,1,?>),
1076// while the latter introduces an edge from the NW to SE corner
1077// (i.e. from <0,1,?> to <1,0,?>.) For a "lazy mesh" look, use
1078// exclusively _03 or _12. For a "classic mesh" look, alternate
1079// _03 and _12 in a checkerboard pattern. For a "smoothest surface"
1080// look, choose the edge based on actual vertex heights.
1081//
1082// The four vertex heights can come from several places. The simplest
1083// encoding is to just use the 'vheight' parameter which stores four
1084// explicit vertex heights for every block. This allows total independence,
1085// but at the cost of the largest memory usage, 1 byte per 3D block.
1086// Encode this with STBVOX_MAKE_VHEIGHT(vh_sw, vh_se, vh_nw, vh_ne).
1087// These coordinates are absolute, not affected by block rotations.
1088//
1089// An alternative if you just want to encode some very specific block
1090// types, not all the possibilities--say you just want half-height slopes,
1091// so you want (0,0,1,1) and (1,1,2,2)--then you can use block_vheight
1092// to specify them. The geometry rotation will cause block_vheight values
1093// to be rotated (because it's as if you're just defining a type of
1094// block). This value is also encoded with STBVOX_MAKE_VHEIGHT.
1095//
1096// If you want to save memory and you're creating a "continuous ground"
1097// sort of effect, you can make each vertex of the lattice share the
1098// vheight value; that is, two adjacent blocks that share a vertex will
1099// always get the same vheight value for that vertex. Then you need to
1100// store two bits of vheight for every block, which you do by storing it
1101// as part another data structure. Store the south-west vertex's vheight
1102// with the block. You can either use the "geometry" mesh variable (it's
1103// a parameter to STBVOX_MAKE_GEOMETRY) or you can store it in the
1104// "lighting" mesh variable if you defined STBVOX_CONFIG_VHEIGHT_IN_LIGHTING,
1105// using STBVOX_MAKE_LIGHTING_EXT(lighting,vheight).
1106//
1107// Note that if you start with a 2D height map and generate vheight data from
1108// it, you don't necessarily store only one value per (x,y) coordinate,
1109// as the same value may need to be set up at multiple z heights. For
1110// example, if height(8,8) = 13.5, then you want the block at (8,8,13)
1111// to store STBVOX_VERTEX_HEIGHT_half, and this will be used by blocks
1112// at (7,7,13), (8,7,13), (7,8,13), and (8,8,13). However, if you're
1113// allowing steep slopes, it might be the case that you have a block
1114// at (7,7,12) which is supposed to stick up to 13.5; that means
1115// you also need to store STBVOX_VERTEX_HEIGHT_one_and_a_half at (8,8,12).

2052 face_data.tex2 = mm->input.rgb[v_off].g;
2053 face_data.color = mm->input.rgb[v_off].b;

2056 }
2057 #else
2058 unsigned char color_face;
2059
2060 if (mm->input.color)
2061 face_data.color = mm->input.color[v_off];
2062
2063 if (mm->input.block_tex1)
2064 face_data.tex1 = mm->input.block_tex1[bt];
2065 else if (mm->input.block_tex1_face)
2066 face_data.tex1 = mm->input.block_tex1_face[bt][bt_face];

2069
2070 if (mm->input.block_tex2)
2071 face_data.tex2 = mm->input.block_tex2[bt];
2072 else if (mm->input.block_tex2_face)
2073 face_data.tex2 = mm->input.block_tex2_face[bt][bt_face];
2074
2075 if (mm->input.block_color) {
2076 unsigned char mcol = mm->input.block_color[bt];
2077 if (mcol)

2079 } else if (mm->input.block_color_face) {

2083 }

2087 facerot = mm->input.side_texrot[v_off] >> (2 * face);
2088 else if (mm->input.block_side_texrot)

2090 }

2094 unsigned char over = mm->input.overlay[v_off];
2095 if (over) {
2096 if (mm->input.overlay_tex1) {
2097 unsigned char rep1 = mm->input.overlay_tex1[over][over_face];
2098 if (rep1)

2100 }

2104 face_data.tex2 = rep2;
2105 }

2108 if (rep3)

2113 facerot = mm->input.overlay_side_texrot[over] >> (2*over_face);
2114 }
2115 }

2117 if (mm->input.tex2_for_tex1)
2118 face_data.tex2 = mm->input.tex2_for_tex1[face_data.tex1];
2119 if (mm->input.tex2)
2120 face_data.tex2 = mm->input.tex2[v_off];
2121 if (mm->input.tex2_replace) {
2122 if (mm->input.tex2_facemask[v_off] & (1 << face))
2123 face_data.tex2 = mm->input.tex2_replace[v_off];
2124 }
2125
2126 color_face = STBVOX_ROTATE(face, rot.ecolor);
2127 if (mm->input.extended_color) {
2128 unsigned char ec = mm->input.extended_color[v_off];
2129 if (mm->input.ecolor_facemask[ec] & (1 << color_face))
2130 face_data.color = mm->input.ecolor_color[ec];
2131 }
2132
2133 if (mm->input.color2) {

2135 face_data.color = mm->input.color2[v_off];
2136 if (mm->input.color3 && (mm->input.color3_facemask[v_off] & (1 << color_face)))
2137 face_data.color = mm->input.color3[v_off];
2138 }
2139 #endif
2140
2141 face_data.face_info = (normal<<2) + facerot;
2142 return face_data;
2143}
