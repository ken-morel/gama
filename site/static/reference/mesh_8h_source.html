48int gm3_mesh_center(gm3Mesh *m) {

50 return -1;
51 gm3Pos max = {DBL_MIN, DBL_MIN, DBL_MIN};
52 gm3Pos min = {DBL_MAX, DBL_MAX, DBL_MAX};
53
54 for (size_t v = 0; v < m->n_vertices; v++) {
55 gm3Pos p = m->vertices[v];
56 min.x = fmin(p.x, min.x);
57 max.x = fmax(p.x, max.x);

59 min.y = fmin(p.y, min.y);
60 max.y = fmax(p.y, max.y);
61

63 max.z = fmax(p.z, max.z);
64 }

67 gm3_pos_center(&center, &max, &min);

70 gm3_pos_substract(&m->vertices[v], &center);
